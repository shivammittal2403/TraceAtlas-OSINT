def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def new_id(prefix: str, seed: str) -> str:
    h = hashlib.sha256(seed.encode("utf-8")).hexdigest()[:10]
    return f"{prefix}{h}" if prefix else h


def stable_hash(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def sha256_short(text: str) -> str:
    return stable_hash(text)[:16]


def jsonable(obj: Any) -> Any:
    if isinstance(obj, Enum):
        return obj.value
    if is_dataclass(obj) and not isinstance(obj, type):
        return {f.name: jsonable(getattr(obj, f.name)) for f in fields(obj)}
    if isinstance(obj, (list, tuple, set)):
        return [jsonable(x) for x in obj]
    if isinstance(obj, dict):
        return {str(k): jsonable(v) for k, v in obj.items()}
    return obj


def normalize_text(value: str) -> str:
    s = unicodedata.normalize("NFKC", value or "")
    s = s.lower().strip()
    s = re.sub(r"[^\w\s\-'.:/@]", " ", s)
    s = re.sub(r"\s+", " ", s)
    return s


def parse_dt(value: Optional[str]) -> Optional[datetime]:
    if not value:
        return None
    try:
        return datetime.fromisoformat(value.replace("Z", "+00:00"))
    except Exception:
        return None


def dt_or_min(value: Optional[str]) -> datetime:
    dt = parse_dt(value)
    return dt if dt else datetime.min.replace(tzinfo=timezone.utc)


def dt_or_max(value: Optional[str]) -> datetime:
    dt = parse_dt(value)
    return dt if dt else datetime.max.replace(tzinfo=timezone.utc)


def overlaps(
    start_a: Optional[str],
    end_a: Optional[str],
    start_b: Optional[str],
    end_b: Optional[str],
) -> bool:
    a_s = parse_dt(start_a)
    a_e = parse_dt(end_a) or a_s
    b_s = parse_dt(start_b)
    b_e = parse_dt(end_b) or b_s

    if not a_s or not b_s:
        return True

    if a_e and b_s and a_e < b_s:
        return False
    if b_e and a_s and b_e < a_s:
        return False
    return True


def policy_guard(text: str) -> List[Dict[str, str]]:
    violations: List[Dict[str, str]] = []
    for rule, rx in PROHIBITED_PATTERNS:
        m = rx.search(text or "")
        if m:
            violations.append({"rule": rule, "matched": m.group(0)})
    return violations

    # -----------------------------------------------------------------
    # Source lineage / independence
    # -----------------------------------------------------------------

    def get_source_family(self, source_id: str) -> Optional[str]:
        src = self.sources.get(source_id)
        if not src:
            return None
        seen: Set[str] = set()
        cur = src
        while (
            cur
            and cur.derived_from
            and cur.derived_from in self.sources
            and cur.id not in seen
        ):
            seen.add(cur.id)
            cur = self.sources[cur.derived_from]
        return cur.id if cur else source_id

    def source_families(self, source_ids: List[str]) -> Set[str]:
        families: Set[str] = set()
        for sid in source_ids:
            fam = self.get_source_family(sid)
            families.add(fam or sid)
        return families

    def independence_state(self, source_ids: List[str]) -> str:
        if not source_ids:
            return "UNKNOWN"
        families = self.source_families(source_ids)
        if len(source_ids) == 1:
            return "SINGLE_SOURCE"
        if len(families) == 1:
            return "DEPENDENT"
        if len(families) == len(source_ids):
            return "INDEPENDENT"
        return "PARTIALLY_DEPENDENT"

    # -----------------------------------------------------------------
    # Temporal configuration
    # -----------------------------------------------------------------

    def config_at(
        self,
        resource_id: str,
        timestamp: Optional[str],
        config_type: Optional[str] = None,
    ) -> Optional[ConfigSnapshot]:
        ts = parse_dt(timestamp) or datetime.now(timezone.utc)
        candidates: List[ConfigSnapshot] = []
        for snap in self.config_snapshots.values():
            if snap.resource_id != resource_id:
                continue
            if config_type and snap.config_type != config_type:
                continue
            vf = parse_dt(snap.valid_from)
            vt = parse_dt(snap.valid_to)
            if vf and vf > ts:
                continue
            if vt and vt < ts:
                continue
            candidates.append(snap)

        if not candidates:
            return None

        return max(candidates, key=lambda c: dt_or_min(c.valid_from))

    def config_history(self, resource_id: str, config_type: Optional[str] = None) -> List[ConfigSnapshot]:
        items = [
            s for s in self.config_snapshots.values()
            if s.resource_id == resource_id and (not config_type or s.config_type == config_type)
        ]
        return sorted(items, key=lambda s: dt_or_min(s.valid_from))

    # -----------------------------------------------------------------
    # IAM helpers (defensive, non-operational)
    # -----------------------------------------------------------------

    def identity_has_role(self, identity_id: str, role_id: str, at: Optional[str] = None) -> bool:
        for att in self.role_attachments.values():
            if att.identity_id == identity_id and att.role_id == role_id:
                if overlaps(att.valid_from, att.valid_to, at, at):
                    return True
        return False

    def trust_exists(self, identity_id: str, role_id: str, at: Optional[str] = None) -> bool:
        for tr in self.trust_relationships.values():
            if tr.from_identity_id == identity_id and tr.to_role_id == role_id:
                if overlaps(tr.valid_from, tr.valid_to, at, at):
                    return True
        return False

    def role_policies(self, role_id: str) -> List[Policy]:
        role = self.roles.get(role_id)
        if not role:
            return []
        return [self.policies[pid] for pid in role.attached_policy_ids if pid in self.policies]

    def policy_allows_action_on_resource(self, policy: Policy, action: str, resource_id: str) -> bool:
        # Very conservative deterministic substring/action matching for demo.
        # Real implementation must use provider-specific IAM policy evaluator.
        for stmt in policy.statements:
            if stmt.effect.lower() != "allow":
                continue
            action_match = any(a == "*" or a.endswith(f":{action}") or action in a for a in stmt.actions)
            resource_match = any(
                r == "*" or resource_id.lower() in r.lower() or normalize_text(resource_id) in normalize_text(r)
                for r in stmt.resources
            )
            if action_match and resource_match:
                return True
        return False

    def has_wildcard_admin_pattern(self, policy: Policy) -> bool:
        for stmt in policy.statements:
            if stmt.effect.lower() != "allow":
                continue
            if "*" in stmt.actions and "*" in stmt.resources:
                return True
        return False

    # -----------------------------------------------------------------
    # Network helpers
    # -----------------------------------------------------------------

    def internet_ingress_allowed(self, resource_id: str, at: Optional[str] = None) -> bool:
        # Demo heuristic: endpoint public + SG rule allows 0.0.0.0/0 inbound.
        for ep in self.network_endpoints.values():
            if ep.resource_id != resource_id and ep.load_balancer_id != resource_id:
                continue
            if not ep.public:
                continue
            if not overlaps(ep.valid_from, ep.valid_to, at, at):
                continue
            for rule in self.security_group_rules.values():
                if rule.direction.lower() != "ingress":
                    continue
                if rule.action.upper() != "ALLOW":
                    continue
                if rule.source_cidr in {"0.0.0.0/0", "::/0"}:
                    if overlaps(rule.valid_from, rule.valid_to, at, at):
                        return True
        return False

    # -----------------------------------------------------------------
    # Evidence helpers
    # -----------------------------------------------------------------

    def has_successful_anonymous_data_plane_event(self, resource_id: str) -> bool:
        for ev in self.audit_events.values():
            if ev.resource_id != resource_id:
                continue
            if not ev.data_plane:
                continue
            if ev.auth_type and ev.auth_type.upper() in {"ANONYMOUS", "UNAUTHENTICATED", "PUBLIC"}:
                if ev.status and ev.status.upper() in {"SUCCESS", "SUCCESSFUL", "200", "ALLOW", "ALLOWED"}:
                    return True
        return False

    def has_data_classification_metadata(self, resource_id: str) -> bool:
        res = self.resources.get(resource_id)
        if not res:
            return False
        return bool(res.data_classification_candidate)

    def has_approved_public_exception(self, resource_id: str) -> bool:
        res = self.resources.get(resource_id)
        if not res:
            return False
        return res.tags.get("public_access_exception") == "APPROVED"

    # -----------------------------------------------------------------
    # Contradictions
    # -----------------------------------------------------------------

    def detect_contradictions(self) -> None:
        existing = {c.description for c in self.contradictions}

        for cspm in self.cspm_findings.values():
            statement_norm = normalize_text(cspm.statement)
            claims_public = "public" in statement_norm or "anonymous" in statement_norm
            if not claims_public:
                continue

            current_cfg = self.config_at(cspm.resource_id, cspm.retrieved_at, "STORAGE_ACCESS")
            if not current_cfg:
                continue

            current_public = bool(current_cfg.attributes.get("public_read"))
            if current_public:
                continue

            desc = (
                f"CSPM finding {cspm.id} reports public/anonymous access for {cspm.resource_id}, "
                f"but native configuration effective at retrieval time {cspm.retrieved_at} indicates private access. "
                "This is consistent with a stale CSPM result or configuration change after scanner evaluation."
            )
            if desc in existing:
                continue

            self.contradictions.append(Contradiction(
                id=new_id("CON-", desc),
                contradiction_type="CSPM_STALE_VS_NATIVE_CONFIG",
                description=desc,
                resource_ids=[cspm.resource_id],
                finding_ids=[fid for fid, f in self.findings.items() if f.subject_id == cspm.resource_id and f.finding_type == FindingType.CSPM_STALE],
                source_ids=sorted({cspm.source_id, current_cfg.source_id}),
                severity="MEDIUM",
                status="OPEN",
                recommended_resolution=(
                    "Use native configuration history as authoritative for temporal state; treat CSPM as derived/stale "
                    "unless independently re-evaluated."
                ),
            ))
            existing.add(desc)

        # Owner tag vs technical owner mismatch, if present.
        for res in self.resources.values():
            tag_owner = res.tags.get("owner")
            tech_owner = res.technical_owner
            if tag_owner and tech_owner and normalize_text(tag_owner) != normalize_text(tech_owner):
                desc = (
                    f"Resource {res.id} has owner tag '{tag_owner}' but authorized inventory technical owner "
                    f"'{tech_owner}'. Tags are metadata claims and do not independently prove ownership."
                )
                if desc in existing:
                    continue
                self.contradictions.append(Contradiction(
                    id=new_id("CON-", desc),
                    contradiction_type="OWNER_TAG_CONFLICT",
                    description=desc,
                    resource_ids=[res.id],
                    source_ids=sorted(set(res.source_ids)),
                    severity="LOW",
                    status="OPEN",
                    recommended_resolution="Resolve owner through authorized CMDB/inventory/process records, not tag alone.",
                ))
                existing.add(desc)

    # -----------------------------------------------------------------
    # Fact gate
    # -----------------------------------------------------------------

    def fact_gate_findings(self) -> None:
        for f in self.findings.values():
            srcs = [self.sources[sid] for sid in f.source_ids if sid in self.sources]
            if not srcs:
                f.confidence = 0.0
                f.verification_state = VerificationState.UNSUPPORTED
                f.limitations.append("No mapped source.")
                continue

            families = self.source_families(f.source_ids)
            max_rel = max((s.reliability for s in srcs), default=0.5)
            direct = max((DIRECT_SOURCE_FACTOR.get(s.source_type, 0.65) for s in srcs), default=0.65)
            base = max_rel * direct

            if len(families) >= 2:
                base = min(0.99, base * 1.05)
            elif len(families) == 1 and len(f.source_ids) > 1:
                base *= 0.90
                f.limitations.append("Multiple sources share one upstream source family.")

            # Semantic guards.
            if f.finding_type == FindingType.RESOURCE_EXISTS:
                if base >= 0.70:
                    f.verification_state = VerificationState.SUPPORTED
                else:
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED
                f.limitations.append("Resource identity is anchored to provider-native resource ID, not display name.")

            elif f.finding_type in {
                FindingType.CONFIGURATION_STATE,
                FindingType.RESOURCE_STATE,
            }:
                if base >= 0.75:
                    f.verification_state = VerificationState.SUPPORTED
                else:
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED
                f.limitations.append("Configuration state is time-bound; current state must not contaminate historical analysis.")

            elif f.finding_type == FindingType.PUBLIC_ACCESS_CONFIGURATION_CANDIDATE:
                cfg = self.config_at(f.subject_id, f.effective_at or f.valid_to, "STORAGE_ACCESS")
                if cfg and cfg.attributes.get("public_read"):
                    f.verification_state = VerificationState.SUPPORTED if base >= 0.72 else VerificationState.PARTIALLY_SUPPORTED
                else:
                    f.verification_state = VerificationState.INCONCLUSIVE
                f.limitations.extend([
                    "Public-access configuration candidate does not prove objects are publicly readable.",
                    "Public-access configuration candidate does not prove sensitive data exposure.",
                    "No bucket listing, object download, or anonymous access test was performed.",
                ])

            elif f.finding_type == FindingType.SENSITIVE_DATA_EXPOSURE:
                has_dp = self.has_successful_anonymous_data_plane_event(f.subject_id)
                has_class = self.has_data_classification_metadata(f.subject_id)
                if has_dp and has_class and base >= 0.75:
                    f.verification_state = VerificationState.SUPPORTED
                elif has_dp or has_class:
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED
                    base = min(base, 0.55)
                else:
                    f.verification_state = VerificationState.INCONCLUSIVE
                    base = min(base, 0.35)
                f.limitations.extend([
                    "Sensitive data exposure requires data-plane access evidence and data classification evidence.",
                    "Public endpoint or public configuration alone is not sensitive data exposure.",
                    "No private object inspection or bucket abuse was performed.",
                ])

            elif f.finding_type == FindingType.ANONYMOUS_ACCESS_SUPPORTED:
                has_dp = self.has_successful_anonymous_data_plane_event(f.subject_id)
                if has_dp and base >= 0.75:
                    f.verification_state = VerificationState.SUPPORTED
                else:
                    f.verification_state = VerificationState.INCONCLUSIVE
                    base = min(base, 0.40)
                f.limitations.append("Anonymous access requires successful data-plane evidence; configuration alone is insufficient.")

            elif f.finding_type == FindingType.PUBLIC_ENDPOINT_CANDIDATE:
                if self.internet_ingress_allowed(f.subject_id, f.effective_at):
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.60 else VerificationState.INCONCLUSIVE
                else:
                    f.verification_state = VerificationState.INCONCLUSIVE
                f.limitations.extend([
                    "Public endpoint candidate does not prove a service is listening.",
                    "Public endpoint does not prove anonymous access or public data.",
                    "No active probing was performed.",
                ])

            elif f.finding_type == FindingType.PUBLIC_SERVICE_CONFIRMED:
                f.verification_state = VerificationState.INCONCLUSIVE
                base = min(base, 0.45)
                f.limitations.append("Public service confirmation requires authorized service/banner/auth evidence; not present in passive demo.")

            elif f.finding_type == FindingType.IAM_POLICY_OBSERVED:
                f.verification_state = VerificationState.SUPPORTED if base >= 0.75 else VerificationState.PARTIALLY_SUPPORTED
                f.limitations.append("Observed IAM policy statement is not effective access by itself.")

            elif f.finding_type == FindingType.IAM_EFFECTIVE_ACCESS_CANDIDATE:
                f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.60 else VerificationState.INCONCLUSIVE
                base = min(base, 0.72)
                f.limitations.extend([
                    "Effective permission depends on identity policy, resource policy, permissions boundary, session policy, conditions, and network controls.",
                    "IAM policy observation is not effective access proof.",
                    "No role assumption or credential test was performed.",
                ])

            elif f.finding_type == FindingType.PRIVILEGED_IDENTITY_CANDIDATE:
                f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.60 else VerificationState.INCONCLUSIVE
                base = min(base, 0.68)
                f.limitations.extend([
                    "Role name 'admin' does not prove administrator privilege.",
                    "Wildcard policy may be constrained by conditions, boundaries, or resource scope.",
                    "No privilege escalation path was operationalized.",
                ])

            elif f.finding_type == FindingType.TRUST_RELATIONSHIP_OBSERVED:
                f.verification_state = VerificationState.SUPPORTED if base >= 0.75 else VerificationState.PARTIALLY_SUPPORTED
                f.limitations.append("Trust relationship does not prove successful role assumption or effective access.")

            elif f.finding_type in {
                FindingType.MISCONFIGURATION_CANDIDATE,
                FindingType.MISCONFIGURATION_SUPPORTED,
                FindingType.BENCHMARK_DEVIATION,
            }:
                if self.has_approved_public_exception(f.subject_id):
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED
                    f.limitations.append("Authorized exception may explain deviation; accepted risk should not be marked fixed.")
                elif f.finding_type == FindingType.BENCHMARK_DEVIATION:
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.65 else VerificationState.INCONCLUSIVE
                else:
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.68 else VerificationState.INCONCLUSIVE
                f.limitations.extend([
                    "Benchmark deviation is not automatically exploitable vulnerability.",
                    "Misconfiguration is not breach without unauthorized access/disclosure evidence.",
                ])

            elif f.finding_type == FindingType.CSPM_FINDING_REPORTED:
                f.verification_state = VerificationState.REPORTED
                base = min(base, 0.65)
                f.limitations.append("CSPM finding is scanner-reported evidence and may be derived from native configuration.")

            elif f.finding_type == FindingType.CSPM_STALE:
                contradiction_present = any(
                    f.subject_id in c.resource_ids and c.contradiction_type == "CSPM_STALE_VS_NATIVE_CONFIG"
                    for c in self.contradictions
                )
                if contradiction_present:
                    f.verification_state = VerificationState.SUPPORTED
                else:
                    f.verification_state = VerificationState.INCONCLUSIVE
                f.limitations.append("Staleness determination uses native configuration history; scanner cache may differ.")

            elif f.finding_type == FindingType.LOGGING_GAP:
                cfg = self.config_at(f.subject_id, f.effective_at or f.valid_to, "STORAGE_ACCESS")
                if cfg and cfg.attributes.get("access_logging_enabled") is False:
                    f.verification_state = VerificationState.SUPPORTED
                else:
                    f.verification_state = VerificationState.INCONCLUSIVE
                f.limitations.append("Absence of logs is not absence of events; coverage gap must be reported.")

            elif f.finding_type == FindingType.BACKUP_RISK:
                f.verification_state = VerificationState.INCONCLUSIVE
                base = min(base, 0.55)
                f.limitations.append("Backup existence does not prove restorability; restore-test evidence required.")

            elif f.finding_type == FindingType.ENCRYPTION_CONTEXT:
                f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.70 else VerificationState.INCONCLUSIVE
                f.limitations.append("Encryption enabled does not equal zero exposure; key scope and identity controls matter.")

            elif f.finding_type == FindingType.VULNERABILITY_CONTEXT:
                f.verification_state = VerificationState.INCONCLUSIVE
                base = min(base, 0.55)
                f.specialist_handoff = f.specialist_handoff or "VULNINT"
                f.limitations.append("CVE association does not prove cloud resource vulnerability or exploitability without version/config/reachability evidence.")

            elif f.finding_type == FindingType.SUPPLY_CHAIN_CONTEXT:
                f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.65 else VerificationState.INCONCLUSIVE
                f.specialist_handoff = f.specialist_handoff or "SUPPLYCHAININT"
                f.limitations.append("Dependency mapping is not compromise evidence.")

            elif f.finding_type == FindingType.INCIDENT_SIGNAL:
                f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.65 else VerificationState.INCONCLUSIVE
                f.specialist_handoff = f.specialist_handoff or "INCIDENTINT"
                f.limitations.append("Cloud alert or suspicious API pattern is not confirmed incident.")

            elif f.finding_type == FindingType.BREACH_CONTEXT:
                f.verification_state = VerificationState.INCONCLUSIVE
                base = min(base, 0.45)
                f.specialist_handoff = f.specialist_handoff or "BREACHINT"
                f.limitations.append("Breach requires unauthorized access/disclosure evidence; misconfiguration alone is not breach.")

            elif f.finding_type == FindingType.COST_ANOMALY_CONTEXT:
                f.verification_state = VerificationState.PARTIALLY_SUPPORTED if base >= 0.65 else VerificationState.INCONCLUSIVE
                f.limitations.append("Cost anomaly is not compromise; may be deployment, scaling, experiment, or billing effect.")

            elif f.finding_type == FindingType.SHARED_RESPONSIBILITY_CONTEXT:
                f.verification_state = VerificationState.SUPPORTED if base >= 0.75 else VerificationState.PARTIALLY_SUPPORTED
                f.limitations.append("Shared responsibility must be assessed by service model and provider/customer boundary.")

            else:
                if base >= 0.78:
                    f.verification_state = VerificationState.SUPPORTED
                elif base >= 0.60:
                    f.verification_state = VerificationState.PARTIALLY_SUPPORTED
                elif base >= 0.40:
                    f.verification_state = VerificationState.INCONCLUSIVE
                else:
                    f.verification_state = VerificationState.UNSUPPORTED

            f.confidence = round(max(0.0, min(0.99, base)), 3)

    def build_risk_dimensions(self) -> None:
        def max_state(types: Set[FindingType]) -> str:
            vals = [f.verification_state.value for f in self.findings.values() if f.finding_type in types]
            if VerificationState.SUPPORTED.value in vals:
                return "SUPPORTED"
            if VerificationState.PARTIALLY_SUPPORTED.value in vals:
                return "PARTIALLY_SUPPORTED"
            if VerificationState.INCONCLUSIVE.value in vals:
                return "INCONCLUSIVE"
            if VerificationState.REPORTED.value in vals:
                return "REPORTED"
            return "UNKNOWN"

        self.risk_dimensions = {
            RiskDimension.IDENTITY_RISK.value: max_state({
                FindingType.IAM_EFFECTIVE_ACCESS_CANDIDATE,
                FindingType.PRIVILEGED_IDENTITY_CANDIDATE,
                FindingType.TRUST_RELATIONSHIP_OBSERVED,
            }),
            RiskDimension.NETWORK_EXPOSURE.value: max_state({
                FindingType.PUBLIC_ENDPOINT_CANDIDATE,
                FindingType.PUBLIC_SERVICE_CONFIRMED,
            }),
            RiskDimension.DATA_EXPOSURE.value: max_state({
                FindingType.PUBLIC_ACCESS_CONFIGURATION_CANDIDATE,
                FindingType.SENSITIVE_DATA_EXPOSURE,
                FindingType.ANONYMOUS_ACCESS_SUPPORTED,
            }),
            RiskDimension.CONFIGURATION_RISK.value: max_state({
                FindingType.MISCONFIGURATION_CANDIDATE,
                FindingType.MISCONFIGURATION_SUPPORTED,
                FindingType.BENCHMARK_DEVIATION,
                FindingType.CSPM_STALE,
            }),
            RiskDimension.VULNERABILITY_RISK.value: max_state({FindingType.VULNERABILITY_CONTEXT}),
            RiskDimension.SECRET_RISK.value: "UNKNOWN_NO_SECRET_EXPOSURE_EVIDENCE_IN_DEMO",
            RiskDimension.LOGGING_GAP.value: max_state({FindingType.LOGGING_GAP}),
            RiskDimension.BACKUP_RISK.value: max_state({FindingType.BACKUP_RISK}),
            RiskDimension.RESILIENCE_RISK.value: "UNKNOWN_RESTORE_TEST_UNAVAILABLE",
            RiskDimension.SUPPLY_CHAIN_RISK.value: max_state({FindingType.SUPPLY_CHAIN_CONTEXT}),
            RiskDimension.INCIDENT_RISK.value: max_state({FindingType.INCIDENT_SIGNAL, FindingType.BREACH_CONTEXT}),
            RiskDimension.EVIDENCE_CONFIDENCE.value: "MEDIUM_PASSIVE_AUTHORIZED_CORPUS",
        }

    def build_knowledge_gaps(self) -> None:
        existing = {g.description for g in self.gaps}

        for f in self.findings.values():
            if f.verification_state == VerificationState.INCONCLUSIVE:
                desc = f"Inconclusive finding {f.id}: {f.statement}"
                if desc not in existing:
                    self.gaps.append(KnowledgeGap(
                        id=new_id("GAP-", desc),
                        gap_type=GapType.PUBLIC_EXPOSURE_UNCERTAIN if "public" in normalize_text(f.statement) else GapType.SOURCE_INDEPENDENCE_GAP,
                        description=desc,
                        about_resource_ids=[f.subject_id],
                        about_finding_ids=[f.id],
                        importance="MEDIUM",
                        recommended_source="Authorized native configuration, data-plane logs, or owner/change records",
                        specialist=f.specialist_handoff,
                        expected_information_value=0.65,
                    ))
                    existing.add(desc)

        for res in self.resources.values():
            if not res.business_owner or not res.technical_owner:
                desc = f"Resource {res.id} ownership/business/technical owner unresolved."
                if desc not in existing:
                    self.gaps.append(KnowledgeGap(
                        id=new_id("GAP-OWN-", desc),
                        gap_type=GapType.OWNERSHIP_UNCLEAR,
                        description=desc,
                        about_resource_ids=[res.id],
                        importance="MEDIUM",
                        recommended_source="Authorized CMDB, inventory, change management, or cost allocation records",
                        specialist="ORGINT / FINOPS",
                        expected_information_value=0.55,
                    ))
                    existing.add(desc)

            if not res.data_classification_candidate:
                desc = f"Resource {res.id} data classification unknown."
                if desc not in existing:
                    self.gaps.append(KnowledgeGap(
                        id=new_id("GAP-DATA-", desc),
                        gap_type=GapType.DATA_CLASSIFICATION_UNKNOWN,
                        description=desc,

                        specialist="DATAINT / DLP",

        for cspm in self.cspm_findings.values():
            desc = f"CSPM finding {cspm.id} may be stale or derived from native configuration."

                self.gaps.append(KnowledgeGap(

                    description=desc,
                    about_resource_ids=[cspm.resource_id],
                    importance="MEDIUM",
                    recommended_source="Re-run authorized CSPM scan or compare native config history",
                    specialist="CSPM / CLOUDSEC",
                    expected_information_value=0.60,
                ))
                existing.add(desc)

        if not any(ev.data_plane for ev in self.audit_events.values()):
            desc = "No data-plane storage access logs are present in the authorized corpus."
            if desc not in existing:
                self.gaps.append(KnowledgeGap(
                    id=new_id("GAP-DPLOG-", desc),
                    gap_type=GapType.LOGGING_COVERAGE_MISSING,
                    description=desc,
                    about_resource_ids=list(self.resources.keys()),
                    importance="HIGH",
                    recommended_source="Authorized storage data-plane access logs / object-level audit logs",
                    specialist="LOGINT / CLOUDINT",
                    expected_information_value=0.90,
                ))
                existing.add(desc)

    def dual_ai_review(self) -> Dict[str, Any]:
        issues: List[str] = []

        if self.contradictions:
            issues.append(f"{len(self.contradictions)} contradiction(s) remain open.")

        stale = [f for f in self.findings.values() if f.finding_type == FindingType.CSPM_STALE and f.verification_state == VerificationState.SUPPORTED]
        if stale:
            issues.append("CSPM findings appear stale relative to native configuration history.")

        data_exp = [f for f in self.findings.values() if f.finding_type == FindingType.SENSITIVE_DATA_EXPOSURE]
        if any(f.verification_state == VerificationState.INCONCLUSIVE for f in data_exp):
            issues.append("Sensitive data exposure is inconclusive due absent data-plane access/classification evidence.")

        iam = [f for f in self.findings.values() if f.finding_type in {FindingType.IAM_EFFECTIVE_ACCESS_CANDIDATE, FindingType.PRIVILEGED_IDENTITY_CANDIDATE}]
        if iam:
            issues.append("IAM effective permissions and privileged identity status remain candidate-only.")

        logging_gap = [f for f in self.findings.values() if f.finding_type == FindingType.LOGGING_GAP]
        if logging_gap:
            issues.append("Logging gap prevents confirmation or exclusion of object-level access.")

        dependent = [
            f for f in self.findings.values()
            if self.independence_state(f.source_ids) in {"DEPENDENT", "SINGLE_SOURCE"}
        ]
        if dependent:
            issues.append("Some findings rely on single-source or dependent source families.")

        if any(f.finding_type == FindingType.VULNERABILITY_CONTEXT for f in self.findings.values()):
            issues.append("Vulnerability applicability requires VULNINT handoff; CVE presence is not exploitation.")

        if not issues:
            verdict = "AGREE"
        elif len(issues) <= 5:
            verdict = "PARTIAL_AGREEMENT"
        else:
            verdict = "INSUFFICIENT_EVIDENCE"

        return {
            "primary_cloud_analyst": (
                "Authorized native configuration shows storage public-read capability during the historical window, "
                "later changed to private. CSPM reporting public access at current retrieval is stale/dependent. "
                "Public endpoint candidate exists for load-balanced service, but service listening and anonymous data access are not established. "
                "IAM policy observations exist, but effective permissions remain unresolved."
            ),
            "independent_cloud_skeptic_issues": issues,
            "verdict": verdict,
            "adversarial_checks": [
                "Are provider and tenant confused? No; provider-native tenant/account/resource IDs preserved.",
                "Is resource name treated as identity? No; resource ID anchors used.",
                "Is owner tag treated as verified owner? No; tag conflict preserved.",
                "Is IAM policy treated as effective access? No; candidate only.",
                "Is public endpoint treated as public data? No; separate findings.",
                "Is CSPM treated as independent confirmation? No; derived/stale flagged.",
                "Is current configuration used for historical event? No; temporal config eras used.",
                "Was any credential/bucket/metadata/IAM action performed? No.",
            ],
            "note": "AI agreement is analytical agreement, not independent cloud source corroboration.",
        }

def graph_version_hash(c: CloudInt) -> str:
    seed_obj = {
        "resources": sorted((rid, r.provider, r.tenant_id, r.account_id, r.resource_type) for rid, r in c.resources.items()),
        "configs": sorted((cid, cs.resource_id, cs.config_type, cs.valid_from, cs.valid_to) for cid, cs in c.config_snapshots.items()),
        "findings": sorted((fid, f.finding_type.value, f.subject_id, f.verification_state.value, round(float(f.confidence), 3)) for fid, f in c.findings.items()),
    }
    return sha256_short(json.dumps(jsonable(seed_obj), sort_keys=True))


def build_source_graph(c: CloudInt) -> Dict[str, Any]:
    edges = []
    for src in c.sources.values():
        if src.derived_from:
            edges.append({
                "source": src.derived_from,
                "target": src.id,
                "relationship_type": "DERIVED_FROM",
            })
    return {
        "nodes": [s.id for s in c.sources.values()],
        "edges": edges,
    }


def build_source_dependency_graph(c: CloudInt) -> Dict[str, List[str]]:
    families: Dict[str, List[str]] = defaultdict(list)
    for sid in c.sources:
        fam = c.get_source_family(sid) or "UNKNOWN"
        families[fam].append(sid)
    return {k: sorted(v) for k, v in families.items()}


def build_config_history_payload(c: CloudInt) -> List[Dict[str, Any]]:
    out = []
    for cs in sorted(c.config_snapshots.values(), key=lambda x: (x.resource_id, dt_or_min(x.valid_from))):
        out.append({
            "config_snapshot_id": cs.id,
            "resource_id": cs.resource_id,
            "config_type": cs.config_type,
            "attributes": cs.attributes,
            "valid_from": cs.valid_from,
            "valid_to": cs.valid_to,
            "observed_at": cs.observed_at,
            "source_id": cs.source_id,
            "evidence_id": cs.evidence_id,
            "parser_version": cs.parser_version,
            "confidence": cs.confidence,
            "limitations": cs.limitations,
        })
    return out


def build_iam_graph(c: CloudInt) -> Dict[str, Any]:
    nodes = []
    edges = []

    for ident in c.identities.values():
        nodes.append({"id": ident.id, "type": "Identity", "identity_type": ident.identity_type.value})
    for role in c.roles.values():
        nodes.append({"id": role.id, "type": "Role", "role_name": role.role_name})
    for pol in c.policies.values():
        nodes.append({"id": pol.id, "type": "Policy", "policy_name": pol.policy_name})

    for att in c.role_attachments.values():
        edges.append({
            "source": att.identity_id,
            "target": att.role_id,
            "relationship_type": "ASSIGNED_ROLE",
            "valid_from": att.valid_from,
            "valid_to": att.valid_to,
        })
    for tr in c.trust_relationships.values():
        edges.append({
            "source": tr.from_identity_id,
            "target": tr.to_role_id,
            "relationship_type": "TRUSTS_CAN_ASSUME",
            "conditions": tr.conditions,
            "valid_from": tr.valid_from,
            "valid_to": tr.valid_to,
        })
    for pol in c.policies.values():
        for target in pol.attached_to:
            edges.append({
                "source": pol.id,
                "target": target,
                "relationship_type": "ATTACHED_TO",
            })

    return {"nodes": nodes, "edges": edges}

def run_unconfigured_pipeline(case: Case) -> Dict[str, Any]:
    c = CloudInt(case)

    c.gaps.append(KnowledgeGap(
        id="GAP-NO-EVIDENCE",
        gap_type=GapType.RESOURCE_UNRESOLVED,
        description="No authorized cloud inventory, configuration, IAM, audit, flow, CSPM, or public metadata corpus is configured.",
        importance="HIGH",
        recommended_source="Connect authorized read-only cloud intelligence sources or provide sanitized local JSON evidence corpus.",
        specialist="CLOUDINT",
        expected_information_value=0.95,
    ))

    c.actions = [NextAction(
        id="ACT-CONFIGURE-EVIDENCE",
        description=(
            "Configure authorized passive cloud sources or supply sanitized local evidence. "
            "Do not use/test credentials, assume roles, access buckets, exploit metadata/SSRF, "
            "modify IAM/logs, or cross tenant boundaries."
        ),
        priority=1,
        privacy_impact="LOW",
        expected_gain=0.95,
        specialist=None,
        requires_human_approval=False,
    )]

    c.handoffs = []
    c.recommendations = []
    c.risk_dimensions = {dim.value: "UNKNOWN_NO_EVIDENCE" for dim in RiskDimension}

    dual = {
        "primary_cloud_analyst": "No evidence available.",
        "independent_cloud_skeptic_issues": [
            "No sources configured.",
            "No tenant/resource resolution possible.",
            "No configuration history available.",
            "No IAM/exposure conclusion possible.",
        ],
        "verdict": "INSUFFICIENT_EVIDENCE",
        "note": "AI agreement is not independent cloud source corroboration.",
    }

    summary = (
        "CLOUD UNRESOLVED: No configured evidence corpus. "
        "No tenants, resources, identities, permissions, exposures, or breaches were fabricated. "
        "Provide authorized passive cloud sources or run sample mode."
    )

    result = build_result(c, Status.BLOCKED_CONFIGURATION)
    result["analyst_summary"] = summary
    result["dual_ai_review"] = dual
    return result

    return {

        "objective": case.objective,

def run_pipeline(case: Case) -> Dict[str, Any]:
    text = " ".join(
        [
            case.objective,
            *case.questions,
            *case.resources,
            case.tenant_id or "",
            case.account_id or "",
            case.authorization or "",
            " ".join(case.scope),
        ]
    )
    violations = policy_guard(text)
    if violations:
        return blocked_policy_result(case, violations)

    if case.sample:
        return run_sample_pipeline()

    return run_unconfigured_pipeline(case)

def main() -> None:
    parser = argparse.ArgumentParser(
        description=(
            "TRACEATLAS CLOUDINT local defensive passive-first cloud intelligence pipeline. "
            "Sample mode uses synthetic authorized-cloud-record-style data."
        )
    )
    parser.add_argument("--sample", action="store_true", help="Run built-in synthetic CLOUDINT sample.")
    parser.add_argument("--objective", help="Defensive cloud intelligence objective.")
    parser.add_argument("--question", action="append", default=[], help="Analytic question. Repeatable.")
    parser.add_argument("--provider", help="Cloud provider hint.")
    parser.add_argument("--tenant", help="Tenant ID.")
    parser.add_argument("--account", help="Account ID.")
    parser.add_argument("--subscription", help="Subscription ID.")
    parser.add_argument("--project", help="Project ID.")
    parser.add_argument("--resource", action="append", default=[], help="Resource ID. Repeatable.")
    parser.add_argument("--time-range", help="Time range hint, e.g. 2026-09-01/2026-10-08.")

    args = parser.parse_args()

    if args.sample or not args.objective:
        case = sample_case()
    else:
        case = Case(
            case_id=new_id("CASE-", args.objective),
            task_id=new_id("TASK-", args.objective),
            objective=args.objective,
            questions=args.question,
            provider=args.provider,
            tenant_id=args.tenant,
            account_id=args.account,
            subscription_id=args.subscription,
            project_id=args.project,
            resources=args.resource,
            time_range=args.time_range,
            sample=False,
        )

    result = run_pipeline(case)
    print(json.dumps(jsonable(result), indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()
